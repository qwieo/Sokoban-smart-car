uint8_t lfsr = 0x7F;

float Generate_PRBS_Signal(float mean_val, float amplitude) {
    
    uint8_t bit = ((lfsr >> 6) ^ (lfsr >> 5)) & 1;
    lfsr = ((lfsr << 1) | bit) & 0x7F;
    if (bit) return mean_val + amplitude;
    else     return mean_val - amplitude;
}
